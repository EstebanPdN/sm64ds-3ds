cmake_minimum_required(VERSION 3.20)
project(sm64ds_ctr LANGUAGES C CXX)
if(NINTENDO_3DS)
  # Bare-metal GCC omits GNU-stack metadata by default. All native objects
  # promise a non-executable stack, and C nested-function trampolines are errors.
  add_compile_options(-Wa,--noexecstack -Werror=trampolines)
endif()
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
option(SM64DS_ENABLE_SANITIZERS "Instrument portable host checks with ASan/UBSan" OFF)
if(SM64DS_ENABLE_SANITIZERS)
  if(NINTENDO_3DS)
    message(FATAL_ERROR "SM64DS_ENABLE_SANITIZERS is a host-only option")
  endif()
  add_compile_options(-fsanitize=address,undefined -fno-sanitize-recover=all -fno-omit-frame-pointer)
  add_link_options(-fsanitize=address,undefined)
endif()
include(cmake/IntegratedNative.cmake)
add_library(sm64ds_assets source/core/assets.cpp)
target_include_directories(sm64ds_assets PUBLIC include)
target_compile_options(sm64ds_assets PRIVATE -Wall -Wextra -Wpedantic)
add_library(sm64ds_math source/core/fixed_math.cpp)
target_include_directories(sm64ds_math PUBLIC include)
add_library(sm64ds_game_input source/core/game_input.cpp)
target_include_directories(sm64ds_game_input PUBLIC include)
add_library(sm64ds_game_geometry source/core/game_geometry.cpp)
target_include_directories(sm64ds_game_geometry PUBLIC include)
add_library(sm64ds_native_data source/core/native_data.cpp)
target_include_directories(sm64ds_native_data PUBLIC include)
add_library(sm64ds_game_draw source/core/game_draw.cpp)
target_include_directories(sm64ds_game_draw PUBLIC include)
add_library(sm64ds_models source/core/model.cpp source/vendor/ntr/bmd.cpp
  source/vendor/ntr/gx.cpp source/vendor/ntr/texture.cpp source/vendor/ntr/lz77.cpp)
target_link_libraries(sm64ds_models PUBLIC sm64ds_assets)
add_library(sm64ds_game_textures source/core/game_textures.cpp)
target_link_libraries(sm64ds_game_textures PUBLIC sm64ds_models sm64ds_game_geometry)
add_library(sm64ds_game_data source/game_data/game_data.cpp source/game_data/native_constants.cpp
  source/game_data/relocation.cpp source/game_data/menu_callbacks.cpp)
target_link_libraries(sm64ds_game_data PUBLIC sm64ds_assets)
target_compile_options(sm64ds_game_data PRIVATE -Wall -Wextra -Wpedantic)
add_library(sm64ds_game_files source/ctr/game_files.cpp source/core/narc.cpp)
target_link_libraries(sm64ds_game_files PUBLIC sm64ds_models)
add_library(sm64ds_collision_data source/core/collision_data.cpp)
target_link_libraries(sm64ds_collision_data PUBLIC sm64ds_game_files sm64ds_game_data)
if(NINTENDO_3DS)
  target_sources(sm64ds_game_files PRIVATE source/ctr/game_file_bridge.cpp)
  enable_language(ASM)
  add_subdirectory(source/ctr)
